function getTeacherFiles() { const props = PropertiesService.getUserProperties(); let folderId = props.getProperty("TEACHER_FOLDER_ID"); let sheetId = props.getProperty("TEACHER_SHEET_ID"); let folder, spreadsheet; if (folderId && sheetId) { folder = DriveApp.getFolderById(folderId); spreadsheet = SpreadsheetApp.openById(sheetId); } else { folder = DriveApp.createFolder("تسجيلات الطلاب الصوتية"); spreadsheet = SpreadsheetApp.create("سجل التسجيلات الصوتية"); props.setProperty("TEACHER_FOLDER_ID", folder.getId()); props.setProperty("TEACHER_SHEET_ID", spreadsheet.getId()); const sheet = spreadsheet.getSheets()[0]; sheet.setName("التسجيلات"); sheet.appendRow([ "اسم الطالب", "التاريخ والوقت", "رابط التسجيل" ]); } return { folder, spreadsheet }; } // يفتحها المعلم مرة واحدة لتفعيل الصلاحيات وتجهيز ملفاته function doGet() { try { const files = getTeacherFiles(); return HtmlService.createHtmlOutput(`

تم ربط حسابك بنجاح

أصبح حسابك جاهزًا لاستقبال تسجيلات الطلاب.

فتح مجلد التسجيلات

فتح سجل التسجيلات

`); } catch (e) { return HtmlService.createHtmlOutput( "تعذر تجهيز الحساب. تأكد من تسجيل الدخول والموافقة على الصلاحيات." ); } } // استقبال التسجيل من موقع GitHub الحالي function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (!data.student || !data.audio || !data.mimeType) { throw new Error("بيانات التسجيل غير مكتملة."); } const files = getTeacherFiles(); const bytes = Utilities.base64Decode(data.audio); if (bytes.length > 10 * 1024 * 1024) { throw new Error("حجم التسجيل أكبر من المسموح."); } const mime = data.mimeType; const extension = mime.includes("mp4") ? "mp4" : mime.includes("ogg") ? "ogg" : "webm"; const timestamp = Utilities.formatDate( new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd_HH-mm-ss" ); const safeName = String(data.student) .replace(/[\\/:*?"<>|]/g, "_"); const blob = Utilities.newBlob( bytes, mime, safeName + "_" + timestamp + "." + extension ); const file = files.folder.createFile(blob); files.spreadsheet.getSheets()[0].appendRow([ data.student, new Date(), file.getUrl() ]); return ContentService .createTextOutput(JSON.stringify({ success: true, message: "تم حفظ التسجيل" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: "تعذر حفظ التسجيل" })) .setMimeType(ContentService.MimeType.JSON); } }
